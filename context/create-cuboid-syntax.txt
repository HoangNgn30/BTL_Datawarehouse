-- ============================================================
-- PHẦN 2: INVENTORY CUBE (32 bảng — olap_i_*)
-- Measure: total_quantity_on_hand = SUM(so_luong) của snapshot mới nhất
-- Chiều: Time(4) × Product(2) × Store(4) = 32
-- LƯU Ý: Dùng subquery lấy MAX(time_key) trong khoảng thời gian đó
-- ============================================================

-- ------------------------------------------------------------
-- HELPER: CTE pattern dùng lại cho snapshot logic
-- Mỗi cuboid sẽ tự inline subquery tương ứng
-- ------------------------------------------------------------

-- 0-D (1 bảng)
-- ------------------------------------------------------------

-- [25] olap_i_all — snapshot mới nhất toàn hệ thống
SELECT
    SUM(f.so_luong) AS total_quantity_on_hand
INTO OLAP_QuanLyBanHang.dbo.olap_i_all
FROM DW_Core.dbo.Fact_Kho f
WHERE f.time_key = (
    SELECT MAX(f2.time_key)
    FROM DW_Core.dbo.Fact_Kho f2
    WHERE f2.ch_key = f.ch_key
      AND f2.mh_key = f.mh_key
);

-- ------------------------------------------------------------
-- 1-D (7 bảng)
-- ------------------------------------------------------------

-- [26] olap_i_year
SELECT
    t.nam               AS year,
    SUM(f.so_luong)     AS total_quantity_on_hand
INTO OLAP_QuanLyBanHang.dbo.olap_i_year
FROM DW_Core.dbo.Fact_Kho f
JOIN DW_Core.dbo.Dim_ThoiGian t ON f.time_key = t.time_key
WHERE f.time_key = (
    SELECT MAX(f2.time_key)
    FROM DW_Core.dbo.Fact_Kho f2
    WHERE f2.ch_key = f.ch_key
      AND f2.mh_key = f.mh_key
      AND f2.time_key / 100 = t.nam
)
GROUP BY t.nam;

-- [27] olap_i_quarter
SELECT
    t.nam               AS year,
    t.quy               AS quarter,
    SUM(f.so_luong)     AS total_quantity_on_hand
INTO OLAP_QuanLyBanHang.dbo.olap_i_quarter
FROM DW_Core.dbo.Fact_Kho f
JOIN DW_Core.dbo.Dim_ThoiGian t ON f.time_key = t.time_key
WHERE f.time_key = (
    SELECT MAX(f2.time_key)
    FROM DW_Core.dbo.Fact_Kho f2
    JOIN DW_Core.dbo.Dim_ThoiGian t2 ON f2.time_key = t2.time_key
    WHERE f2.ch_key = f.ch_key
      AND f2.mh_key = f.mh_key
      AND t2.nam = t.nam
      AND t2.quy = t.quy
)
GROUP BY t.nam, t.quy;

-- [28] olap_i_month
SELECT
    t.nam               AS year,
    t.quy               AS quarter,
    t.thang             AS month,
    SUM(f.so_luong)     AS total_quantity_on_hand

FROM DW_Core.dbo.Fact_Kho f
JOIN DW_Core.dbo.Dim_ThoiGian t ON f.time_key = t.time_key
GROUP BY t.nam, t.quy, t.thang;
-- (month level: mỗi tháng chỉ có 1 snapshot → SUM trực tiếp)

-- [29] olap_i_product

FROM DW_Core.dbo.Fact_Kho f
JOIN DW_Core.dbo.Dim_MatHang mh ON f.mh_key = mh.mh_key
WHERE f.time_key = (
    SELECT MAX(f2.time_key)
    FROM DW_Core.dbo.Fact_Kho f2

      AND f2.mh_key = f.mh_key
)
GROUP BY mh.mh_key, mh.mo_ta;
